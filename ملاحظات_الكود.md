##  تنظيف القيم الناقصة
```python
df_clean = dataset.copy()                                         # نسخة من البيانات عشان الأصل ما يتغيّر
df_clean["SavingAccounts"] = df_clean["SavingAccounts"].fillna("none")      # بيعبّي الناقص في حساب التوفير بكلمة "none" (يعني ما في حساب)
df_clean["CheckingAccount"] = df_clean["CheckingAccount"].fillna("none")    # نفس الشي للحساب الجاري
print("Remaining missing values:\n")                              # عنوان
print(df_clean.isna().sum())                                      # تأكيد إنو ما ضل أي قيمة ناقصة (كلها صفر)
```
##  تحويل الكلمات لأرقام
```python
df_encoded = df_clean.copy()                                      # نسخة جديدة للتحويل
categorical_cols_all = ["Sex", "Housing", "SavingAccounts", "CheckingAccount", "Purpose", "Risk"]  # الأعمدة النصية اللي بدنا نحوّلها
encoders = {}                                                     # قاموس فاضي بنحفظ فيه المحوّلات عشان نرجّع الكلمات لاحقاً
for col in categorical_cols_all:                                  # حلقة على كل عمود نصي
    le = LabelEncoder()                                           # بيعمل محوّل جديد
    df_encoded[col] = le.fit_transform(df_clean[col])             # بيعطي كل كلمة رقم (مثلاً female=0، male=1)
    encoders[col] = le                                            # بيحفظ المحوّل بالقاموس
    print(f"{col:16s} -> {list(le.classes_)}")                    # بيطبع الكلمات حسب ترتيب أرقامها
```
## : توحيد المقياس
```python
scaler = StandardScaler()                                         # بيعمل أداة التوحيد
numeric_cols_to_scale = ["Age", "Job", "CreditAmount", "Duration"]   # الأعمدة الرقمية اللي بدنا نوحّد مقياسها
df_scaled = df_encoded.copy()                                     # نسخة جديدة
df_scaled[numeric_cols_to_scale] = scaler.fit_transform(df_encoded[numeric_cols_to_scale])   # بيخلي متوسط كل عمود صفر وانحرافو المعياري 1، عشان الأرقام الكبيرة ما تطغى على الصغيرة
print("Before scaling:")                                          # عنوان
print(df_encoded[numeric_cols_to_scale].head(3))                  # أول 3 صفوف قبل التوحيد
print("\nAfter scaling (mean ~0, std ~1):")                       # عنوان
print(df_scaled[numeric_cols_to_scale].head(3).round(3))          # أول 3 صفوف بعد التوحيد، مقرّبة لـ 3 أرقام عشرية
```

##  ترتيب الأعمدة حسب الأهمية
```python
X = df_scaled.drop(columns=["Risk"])                              # X هي الأعمدة اللي بنستخدمها (كل شي ما عدا Risk)
y = df_scaled["Risk"]                                             # y هو العمود اللي بدنا نتوقعو (Risk)
selector = SelectKBest(score_func=f_classif, k="all")             # أداة بتعطي درجة لكل عمود، وk="all" يعني ما تحذف أي عمود
selector.fit(X, y)                                                # بتحسب الدرجات (بدون تدريب نموذج)
feature_scores = pd.DataFrame({
    "Feature": X.columns,                                         # أسماء الأعمدة
    "Score": selector.scores_                                     # درجة كل عمود (الأعلى = أهم)
}).sort_values("Score", ascending=False).reset_index(drop=True)   # ترتيب من الأعلى للأقل وترقيم جديد
feature_scores                                                    # عرض الجدول (CheckingAccount طلع الأول)
```
##  رسمة الأهمية
```python
plt.figure(figsize=(8, 5))                                        # حجم الرسمة
sns.barplot(data=feature_scores, x="Score", y="Feature", color="#1E2761")   # أعمدة أفقية: كل ما كان العمود أطول، العمود الأصلي أهم
plt.title("Feature scores (SelectKBest, f_classif) vs. Risk")     # عنوان
plt.xlabel("F-score (higher = more informative)")                 # اسم المحور الأفقي
plt.tight_layout()                                                # ترتيب المسافات
plt.show()                                                        # عرض الرسم
```
